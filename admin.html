showPanelState(false); } } async function loadSiteSettings() { try { const settings = await apiRequest('/api/site-settings'); Object.entries(settings).forEach(([key, value]) => { const field = document.getElementById(`site-${key}`); siteStatus.dataset.state = 'success'; siteStatus.textContent = result.message; } catch (error) { siteStatus.dataset.state = 'error'; siteStatus.textContent = error.message; } }); logoutButton.addEventListener('click', async () => { try { await apiRequest('/api/admin/logout', { method: 'POST' }); showPanelState(false); loginStatus.textContent = ''; feedStatus.textContent = ''; siteStatus.textContent = ''; } catch (error) { console.error('Logout failed', error); } }); checkSession();
